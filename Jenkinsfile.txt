pipeline {
    agent any

    tools {
        jdk 'JDK21'
    }

    stages {

        stage('Clonar Repositorio') {
            steps {
                checkout scm
            }
        }

        stage('Compilar Proyecto') {
            steps {
                bat 'mvnw.cmd clean compile'
            }
        }

        stage('Empaquetar Proyecto') {
            steps {
                bat 'mvnw.cmd clean package -DskipTests'
            }
        }

    }

    post {
        success {
            echo 'Pipeline ejecutado correctamente.'
        }

        failure {
            echo 'El pipeline ha fallado.'
        }
    }
}
